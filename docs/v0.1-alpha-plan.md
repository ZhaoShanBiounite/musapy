### 1.3 Success Definition

The following Python code runs end-to-end on MUSA hardware:

```python
import musapy as ms

ms.set_default_device("musa:0")

# Array creation
a = ms.array([1.0, 2.0, 3.0], dtype=ms.float32)
b = ms.array([4.0, 5.0, 6.0], dtype=ms.float32)

# Minimal op (elementwise add)
c = ms.add(a, b)   # → Array([5.0, 7.0, 9.0], device=musa:0)

# Stream context
s = ms.Stream("musa:0")
with ms.stream(s):
    d = ms.add(a, b)   # bound to s

# Resolution feedback
print(a.device)
# Device(musa:0)  # resolved from: global_default (musa:0), set via ms.set_default_device() at <stdin>:3

# Synchronize + readback
s.synchronize()
print(c.tolist())   # [5.0, 7.0, 9.0]

# Memory summary
ms.memory_summary()
# Device: musa:0
# Used by musapy: 48 bytes (2 buffers)
# Cached (pool): 0 bytes
```

## 2. Verification Prerequisites

**Before starting v0.1-alpha implementation**, a colleague with MUSA hardware must verify:

### 2.1 Stream-Ordered Alloc/Free Test (L3-15)

Minimal C test program (not Rust, not Python — pure MUSA runtime):

```c
// verify_stream_ordered.c
#include <musa_runtime.h>
#include <stdio.h>

int main() {
    // Check capability
    int supported = 0;
    musaDeviceGetAttribute(&supported, musaDevAttrMemoryPoolsSupported, 0);
    if (!supported) {
        printf("FAIL: stream-ordered allocator not supported\n");
        return 1;
    }

    // Create two streams
    musaStream_t s1, s2;
    musaStreamCreate(&s1);
    musaStreamCreate(&s2);

    // Alloc on s1
    void* ptr;
    musaMallocAsync(&ptr, 1024, s1);

    // Record write event on s1
    musaEvent_t write_ev;
    musaEventCreate(&write_ev);
    musaEventRecord(write_ev, s1);

    // s2 waits for s1's write
    musaStreamWaitEvent(s2, write_ev, 0);

    // Use on s2 (would be a kernel launch; here just a memcpy as proxy)
    // ... (kernel launch omitted, use musaMemcpyAsync as proxy)

    // Record read event on s2
    musaEvent_t read_ev;
    musaEventCreate(&read_ev);
    musaEventRecord(read_ev, s2);

    // Free on s2 (last-used stream, strategy b)
    // s2 must wait read_ev (trivially true since read recorded on s2)
    musaFreeAsync(ptr, s2);

    // Synchronize
    musaStreamSynchronize(s2);

    printf("PASS: stream-ordered alloc/free works\n");
    return 0;
}
```

### Phase 3 Cargo Feature
**musapy-core/Cargo.toml adds feature:**

```[features]default = []stream-ordered = []  # enable musaMallocAsync/musaFreeAsync path (requires 5.x SDK)```
Build commands:

**3.x/4.x environment:** cargo build (default, deferred-free)
5.x environment: cargo build --features stream-ordered (stream-ordered)

**Code conventions:**
```#[cfg(feature = "stream-ordered")] gates all async API declarations and calls
Buffer::alloc/Drop selects path based on feature internally
deferred_free.rs only compiled under default path, skipped under stream-ordered feature
runtime probe (probe_memory_pools_supported) as double safety — even with stream-orderedfeature compiled, probe at startup; if unsupported, fallback to deferred-free
```



### Acceptance
**Default path (deferred-free, 3.x/4.x/5.x)：**

```let dev = Device::Musa(0);let stream = Stream::new(dev, 0)?;let buf = Buffer::alloc(1024, dev, &stream)?;buf.record_write(&stream);// ... use buffer ...drop(buf);  // → enqueue to deferred-free queue, no immediate freeStream::synchronize(&stream)?;  // → batch reclaim, musaFree
stream-ordered path (--features stream-ordered, 5.x only)：

let dev = Device::Musa(0);let stream = Stream::new(dev, 0)?;let buf = Buffer::alloc(1024, dev, &stream)?;  // → musaMallocAsyncbuf.record_write(&stream);// ... use buffer ...drop(buf);  // → musaFreeAsync, stream-ordered freeStream::synchronize(&stream)?;  // no reclaim needed (free is stream-ordered)
Both paths have identical external behavior: user code (alloc/record_write/drop/synchronize) is exactly the same, differences are internal only. Feature toggle doesnot affect API.
```

### Kernel Code (`kernels/elementwise.mu`)

```cpp
#include "include/common.h"

template <typename T>
__global__ void musapy_add_kernel(
    const T* __restrict__ a,
    const T* __restrict__ b,
    T* __restrict__ c,
    size_t n
) {
    size_t idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < n) {
        c[idx] = a[idx] + b[idx];
    }
}

extern "C" {
    void musapy_add_f32_v1(
        const float* a, const float* b, float* c,
        size_t n, musaStream_t stream
    ) {
        musapy_add_kernel<float><<<grid_size_1d(n), 256, 0, stream>>>(a, b, c, n);
    }

    void musapy_add_f64_v1(
        const double* a, const double* b, double* c,
        size_t n, musaStream_t stream
    ) {
        musapy_add_kernel<double><<<grid_size_1d(n), 256, 0, stream>>>(a, b, c, n);
    }
}
```
